176subroutine ddrun(ddx_data, state, electrostatics, psi, tol, esolv, &
177 & ddx_error, force, read_guess)
178 type(ddx_type), intent(inout) :: ddx_data
179 type(ddx_state_type), intent(inout) :: state
180 type(ddx_electrostatics_type), intent(in) :: electrostatics
181 real(dp), intent(in) :: psi(ddx_data % constants % nbasis, &
182 & ddx_data % params % nsph)
183 real(dp), intent(in) :: tol
184 real(dp), intent(out) :: esolv
185 type(ddx_error_type), intent(inout) :: ddx_error
186 real(dp), intent(out), optional :: force(3, ddx_data % params % nsph)
187 logical, intent(in), optional :: read_guess
188 ! local variables
189 logical :: do_guess
190
191 ! decide if the guess has to be read or must be done
192 if (present(read_guess)) then
193 do_guess = .not.read_guess
194 else
195 do_guess = .true.
196 end if
197
198 ! if the forces are to be computed, but the array is not passed, raise
199 ! an error
200 if ((.not.present(force)) .and. (ddx_data % params % force .eq. 1)) then
201 call update_error(ddx_error, &
202 & "ddrun: forces are to be computed, but the optional force" // &
203 & " array has not been passed.")
204 return
205 end if
206
207 call setup(ddx_data % params, ddx_data % constants, &
208 & ddx_data % workspace, state, electrostatics, psi, ddx_error)
209 if (ddx_error % flag .ne. 0) then
210 call update_error(ddx_error, "ddrun: setup returned an error, exiting")
211 return
212 end if
213
214 ! solve the primal linear system
215 if (do_guess) then
216 call fill_guess(ddx_data % params, ddx_data % constants, &
217 & ddx_data % workspace, state, tol, ddx_error)
218 if (ddx_error % flag .ne. 0) then
219 call update_error(ddx_error, &
220 & "ddrun: fill_guess returned an error, exiting")
221 return
222 end if
223 end if
224 call solve(ddx_data % params, ddx_data % constants, &
225 & ddx_data % workspace, state, tol, ddx_error)
226 if (ddx_error % flag .ne. 0) then
227 call update_error(ddx_error, "ddrun: solve returned an error, exiting")
228 return
229 end if
230
231 ! compute the energy
232 call energy(ddx_data % params, ddx_data % constants, &
233 & ddx_data % workspace, state, esolv, ddx_error)
234 if (ddx_error % flag .ne. 0) then
235 call update_error(ddx_error, "ddrun: energy returned an error, exiting")
236 return
237 end if
238
239 ! solve the primal linear system
240 if (ddx_data % params % force .eq. 1) then
241 if (do_guess) then
242 call fill_guess_adjoint(ddx_data % params, ddx_data % constants, &
243 & ddx_data % workspace, state, tol, ddx_error)
244 if (ddx_error % flag .ne. 0) then
245 call update_error(ddx_error, &
246 & "ddrun: fill_guess_adjoint returned an error, exiting")
247 return
248 end if
249 end if
250 call solve_adjoint(ddx_data % params, ddx_data % constants, &
251 & ddx_data % workspace, state, tol, ddx_error)
252 if (ddx_error % flag .ne. 0) then
253 call update_error(ddx_error, &
254 & "ddrun: solve_adjoint returned an error, exiting")
255 return
256 end if
257 end if
258
259 ! compute the forces
260 if (ddx_data % params % force .eq. 1) then
261 force = zero
262 call solvation_force_terms(ddx_data % params, ddx_data % constants, &
263 & ddx_data % workspace, state, electrostatics, force, ddx_error)
264 if (ddx_error % flag .ne. 0) then
265 call update_error(ddx_error, &
266 & "ddrun: solvation_force_terms returned an error, exiting")
267 return
268 end if
269 end if
270
271end subroutine ddrun
