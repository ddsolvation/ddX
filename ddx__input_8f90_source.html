122 if (line(k:k) .eq. achar(9)) line(k:k) = " "

230 end do
231 close(unit)
232 if (ddx_error % flag .ne. 0) return
233
234 if (.not. have_model) &
235 & call update_error(ddx_error, "missing key `model`")
236 if (have_model .and. (model .lt. 1 .or. model .gt. 3)) &
237 & call update_error(ddx_error, "`model` must be cosmo," // &
238 & " pcm, lpb or 1..3")

240 & call update_error(ddx_error, "missing key `eps`")

247 end if
248 if (nproc .lt. 0) &
249 & call update_error(ddx_error, "`nproc` must be non-negative")

251 & call update_error(ddx_error, "`lmax` must be non-negative")
252 if (ngrid .lt. 0) &
253 & call update_error(ddx_error, "`ngrid` must be non-negative")
254 if (have_eps .and. eps .lt. zero) &
255 & call update_error(ddx_error, "`eps` must be non-negative")

274 & call update_error(ddx_error, "`fmm` must be 0 or 1")

405 & "file " // trim(fname) // ": `maxiter` must be a positive " // &

407 end if

419 & "file " // trim(fname) // ": `force` must be an integer " // &
420 "value of a value 0 or 1.")
421 end if

426 & "file " // trim(fname) // ": `fmm` must be an integer " // &
427 & "value of a value 0 or 1.")
428 end if

433 & "file " // trim(fname) // ": `pm` must be a non-negative " // &
434 & "integer value.")
435 end if

442 end if

444 read(100, *) nsph
